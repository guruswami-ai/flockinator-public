cmake_minimum_required(VERSION 3.16)
project(flock_management_response_tests LANGUAGES C CXX)
set(CMAKE_C_STANDARD 99)
set(CMAKE_CXX_STANDARD 17)
set(MBEDTLS_SOURCE_DIR "" CACHE PATH "Already-installed upstream Mbed TLS source directory")
if(NOT EXISTS "${MBEDTLS_SOURCE_DIR}/library/md.c")
    message(FATAL_ERROR "Set MBEDTLS_SOURCE_DIR to an existing upstream Mbed TLS source directory")
endif()
add_library(portable_mbedtls STATIC
    "${MBEDTLS_SOURCE_DIR}/library/md.c"
    "${MBEDTLS_SOURCE_DIR}/library/sha256.c"
    "${MBEDTLS_SOURCE_DIR}/library/platform_util.c")
target_include_directories(portable_mbedtls PUBLIC
    "${MBEDTLS_SOURCE_DIR}/include" "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(portable_mbedtls PUBLIC MBEDTLS_CONFIG_FILE="host_mbedtls_config.h")
add_executable(test_protocol test_protocol.cpp)
target_compile_options(test_protocol PRIVATE -Wall -Wextra -Werror)
target_link_libraries(test_protocol PRIVATE portable_mbedtls)
enable_testing()
add_test(NAME management_response_protocol COMMAND test_protocol)
set(GENERATED_PRODUCTION_DIR "" CACHE PATH "Production excerpts generated by run_native.py")
if(GENERATED_PRODUCTION_DIR)
    add_executable(test_production test_production.cpp)
    target_compile_options(test_production PRIVATE -Wall -Wextra -Werror)
    target_include_directories(test_production PRIVATE "${GENERATED_PRODUCTION_DIR}"
        "${CMAKE_CURRENT_SOURCE_DIR}/../.."
        "${CMAKE_CURRENT_SOURCE_DIR}/../../../node-platform/include"
        "${CMAKE_CURRENT_SOURCE_DIR}/../../../korvo-monitor/main")
    target_link_libraries(test_production PRIVATE portable_mbedtls)
    add_test(NAME management_response_production_http COMMAND test_production)
endif()
